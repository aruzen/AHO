    cmake_policy(SET CMP0167 NEW)

message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")

if(NOT DEFINED AHO_HOME)
    if(DEFINED ENV{AHO_HOME} AND NOT "$ENV{AHO_HOME}" STREQUAL "")
        set(AHO_HOME "$ENV{AHO_HOME}" CACHE PATH "AHO library root")
    else()
        set(AHO_HOME "${CMAKE_CURRENT_SOURCE_DIR}/AHOLi/AHO" CACHE PATH "AHO library root")
    endif()
endif ()
if (NOT DEFINED VSL_HOME)
    if(DEFINED ENV{VSL_HOME} AND NOT "$ENV{VSL_HOME}" STREQUAL "")
        set(VSL_HOME "$ENV{VSL_HOME}" CACHE PATH "VSL library root")
    else()
        set(VSL_HOME "${CMAKE_CURRENT_SOURCE_DIR}/VSLi/VSL" CACHE PATH "VSL library root")
    endif()
endif ()
if(WIN32)
    set(VULKAN_SDK "$ENV{VULKAN_SDK}" CACHE PATH "Vulkan SDK root")
    if(DEFINED ENV{GLFW_ROOT})
        set(GLFW_ROOT "$ENV{GLFW_ROOT}" CACHE PATH "GLFW root")
    else()
        set(GLFW_ROOT "$ENV{GLFW}" CACHE PATH "GLFW root (legacy GLFW environment variable)")
    endif()
endif()
message(STATUS "AHO_HOME : ${AHO_HOME}")
message(STATUS "VSL_HOME : ${VSL_HOME}")

project(AHO-all)
string(REPLACE "-fcolor-diagnostics" "-fdiagnostics-color=auto" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")

if(MSVC)
    add_compile_options(/std:c++latest /permissive- /Zc:preprocessor /utf-8)
endif()

# setting Libraries
if(WIN32)
    find_path(Vulkan_INCLUDE_DIR vulkan/vulkan.h HINTS "${VULKAN_SDK}/Include" REQUIRED)
    find_library(Vulkan_LIBRARY NAMES vulkan-1 HINTS "${VULKAN_SDK}/Lib" REQUIRED)
    find_package(Vulkan REQUIRED)
    get_filename_component(Vulkan_LIBRARY_DIRS "${Vulkan_LIBRARY}" DIRECTORY)

    find_path(GLFW_INCLUDE_DIRS GLFW/glfw3.h HINTS "${GLFW_ROOT}/include" REQUIRED)
    option(AHO_GLFW_SHARED "Use the GLFW DLL to avoid mixing Debug and Release CRTs" ON)
    if(AHO_GLFW_SHARED)
        find_library(GLFW_DLL_LIBRARY NAMES glfw3dll HINTS "${GLFW_ROOT}/lib-vc2022" "${GLFW_ROOT}/lib-vc2019" REQUIRED)
        set(GLFW_LIBRARIES "${GLFW_DLL_LIBRARY}")
        add_compile_definitions(GLFW_DLL)
    else()
        find_library(GLFW_LIBRARIES NAMES glfw3 HINTS "${GLFW_ROOT}/lib-vc2022" "${GLFW_ROOT}/lib-vc2019" REQUIRED)
    endif()
    get_filename_component(GLFW_LIBRARY_DIRS "${GLFW_LIBRARIES}" DIRECTORY)
    if(AHO_GLFW_SHARED)
        find_file(GLFW_RUNTIME_LIBRARY glfw3.dll HINTS "${GLFW_LIBRARY_DIRS}" REQUIRED)
    endif()

    if(NOT TARGET glm::glm)
        find_path(GLM_INCLUDE_DIR glm/glm.hpp HINTS "${Vulkan_INCLUDE_DIRS}" REQUIRED)
        add_library(glm::glm INTERFACE IMPORTED)
        target_include_directories(glm::glm INTERFACE "${GLM_INCLUDE_DIR}")
    endif()

    find_package(Boost COMPONENTS system QUIET)
    if(NOT Boost_FOUND)
        set(Boost_INCLUDE_DIRS "")
        set(Boost_LIBRARY_DIRS "")
        set(Boost_LIBRARIES "")
        message(STATUS "Boost not found; continuing because Windows build does not currently use Boost symbols.")
    endif()
else()
    find_package(PkgConfig REQUIRED)
    find_package(Boost COMPONENTS system QUIET)
    if (NOT Boost_FOUND)
        message(STATUS "Boost Config mode failed. Retrying with FindBoost...")
        # add_definitions(-DBOOST_ERROR_CODE_HEADER_ONLY)
        set(Boost_NO_BOOST_CMAKE ON)
        find_package(Boost REQUIRED COMPONENTS system)
    endif ()
    message("Boost: ${Boost_INCLUDE_DIRS}, ${Boost_LIBRARY_DIRS}, ${Boost_LIBRARIES}")

    find_package(Vulkan REQUIRED)
    get_filename_component(Vulkan_LIBRARY_DIRS "${Vulkan_LIBRARIES}" DIRECTORY)
    message("Vulkan: ${Vulkan_INCLUDE_DIRS}, ${Vulkan_LIBRARIES}")
    pkg_check_modules(GLFW REQUIRED glfw3)
    message("glfw3: ${GLFW_INCLUDE_DIRS}, ${GLFW_LIBRARY_DIRS}, ${GLFW_LIBRARIES}")
endif()

message("Vulkan: ${Vulkan_INCLUDE_DIRS}, ${Vulkan_LIBRARIES}")
message("glfw3: ${GLFW_INCLUDE_DIRS}, ${GLFW_LIBRARY_DIRS}, ${GLFW_LIBRARIES}")

function(aho_copy_glfw_runtime target)
    if(WIN32 AND AHO_GLFW_SHARED)
        add_custom_command(TARGET ${target} POST_BUILD
                COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                        "${GLFW_RUNTIME_LIBRARY}" "$<TARGET_FILE_DIR:${target}>"
                VERBATIM
        )
    endif()
endfunction()

include(CTest)
set(AHO_TESTING_HAS_GTEST OFF)
if(BUILD_TESTING)
    set(AHO_GOOGLETEST_ROOT "${CMAKE_SOURCE_DIR}/thirdparty/googletest" CACHE PATH "Local GoogleTest source directory")

    if(EXISTS "${AHO_GOOGLETEST_ROOT}/CMakeLists.txt")
        set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
        add_subdirectory("${AHO_GOOGLETEST_ROOT}" "${CMAKE_BINARY_DIR}/thirdparty/googletest")
        set(AHO_TESTING_HAS_GTEST ON)
    else()
        find_package(GTest QUIET)
        if(GTest_FOUND)
            set(AHO_TESTING_HAS_GTEST ON)
        else()
            message(STATUS "GoogleTest not found. Put it in ${AHO_GOOGLETEST_ROOT} or install GTest to enable tests.")
        endif()
    endif()
endif()

# Debug option
if (CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(CMAKE_CXX_FLAGS_DEBUG "-O0 -g3 -DDEBUG -fstandalone-debug -fno-omit-frame-pointer")
    set(CMAKE_C_FLAGS_DEBUG   "-O0 -g3 -DDEBUG -fstandalone-debug -fno-omit-frame-pointer")
    add_compile_options(-Wno-user-defined-literals)
elseif (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    add_compile_options(-fconcepts-diagnostics-depth=7)
    set(CMAKE_CXX_FLAGS_DEBUG "-O0 -g2 -DDEBUG -gdwarf-4 -fvar-tracking-assignments -fno-omit-frame-pointer")
    set(CMAKE_C_FLAGS_DEBUG   "-O0 -g2 -DDEBUG -gdwarf-4 -fvar-tracking-assignments -fno-omit-frame-pointer")
endif()

# Release option
if (CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3 -march=native -ftree-vectorize")
    set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} -O3 -march=native -ftree-vectorize")
elseif (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3 -march=native -ftree-vectorize -fomit-frame-pointer")
    set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} -O3 -march=native -ftree-vectorize -fomit-frame-pointer")
endif()
